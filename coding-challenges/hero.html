combinationSum.ts LC#39 · Backtracking 
function combinationSum(nums: number[], target: number) {
  const res = [], path = [];
  nums.sort((a, b) => a - b);
  const go = (start, left) => {
    if (left === 0) return res.push([...path]);
    for (let i = start; i < nums.length; i++) {
      if (nums[i] > left) break;   // prune — sorted
      path.push(nums[i]);
      go(i, left - nums[i]);       // i ⇒ reuse
      path.pop();                   // backtrack
    }
  };
  go(0, target);
  return res;
}
